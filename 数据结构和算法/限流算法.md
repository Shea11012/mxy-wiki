# 对比


| 算法   | 优点                       | 缺点                             | 场景                     |
| ---- | ------------------------ | ------------------------------ | ---------------------- |
| 固定窗口 | 适用于稳定的流量控制               | 请求分布不均匀<br>无法应对突发流量<br>存在请求不公平 | 稳定的流量控制，不需要保证请求均匀分布的场景 |
| 滑动窗口 | 灵活、实时性强、精度更高             | 内存消耗大                           | 需要平滑处理突发流量             |
| 漏桶算法 | 平滑突发流量，防止系 统过载            | 突发流量处理不灵活;流出速率固定，造成流量浪费        | 需固定输出速率                |
| 令牌桶  | 平滑流量，允许突发;可动态调整规则，适应流量变化 | 需要维护令牌桶状态                      | 需要动态调整限流规则、平滑处理突发流量    |

>[!note] 分布式限流 
>在分布式 限流中，redis性能高于基于etcd、zk等方案，如果对一致性 要求极高 的场景



# 实现
## 固定窗口限流

```go
type FixedWindowLimiter struct {
   windowSize  time.Duration // 窗口大小
   maxRequests int           // 最大请求数
   requests    int           // 当前窗口内的请求数
   lastReset   int64         // 上次窗口重置时间（秒级时间戳）
   resetMutex  sync.Mutex    // 重置锁
}

func NewFixedWindowLimiter(windowSize time.Duration, maxRequests int) *FixedWindowLimiter {
   return &FixedWindowLimiter{
      windowSize:  windowSize,
      maxRequests: maxRequests,
      lastReset:   time.Now().Unix(),
   }
}

func (limiter *FixedWindowLimiter) AllowRequest() bool {
   limiter.resetMutex.Lock()
   defer limiter.resetMutex.Unlock()

   // 检查是否需要重置窗口
   if time.Now().Unix()-limiter.lastReset >= int64(limiter.windowSize.Seconds()) {
      limiter.requests = 0
      limiter.lastReset = time.Now().Unix()
   }

   // 检查请求数是否超过阈值
   if limiter.requests >= limiter.maxRequests {
      return false
   }

   limiter.requests++
   return true
}

func main() {
   limiter := NewFixedWindowLimiter(1*time.Second, 3) // 每秒最多允许3个请求
   for i := 0; i < 15; i++ {
      now := time.Now().Format("15:04:05")
      if limiter.AllowRequest() {
         fmt.Println(now + " 请求通过")
      } else {
         fmt.Println(now + " 请求被限流")
      }
      time.Sleep(100 * time.Millisecond)
   }
}
```


## 滑动窗口限流
```go
package main

import (
   "fmt"
   "sync"
   "time"
)

type SlidingWindowLimiter struct {
   windowSize   time.Duration // 窗口大小
   maxRequests  int           // 最大请求数
   requests     []time.Time   // 窗口内的请求时间
   requestsLock sync.Mutex    // 请求锁
}

func NewSlidingWindowLimiter(windowSize time.Duration, maxRequests int) *SlidingWindowLimiter {
   return &SlidingWindowLimiter{
      windowSize:  windowSize,
      maxRequests: maxRequests,
      requests:    make([]time.Time, 0),
   }
}

func (limiter *SlidingWindowLimiter) AllowRequest() bool {
   limiter.requestsLock.Lock()
   defer limiter.requestsLock.Unlock()

   // 移除过期的请求
   currentTime := time.Now()
   for len(limiter.requests) > 0 && currentTime.Sub(limiter.requests[0]) > limiter.windowSize {
      limiter.requests = limiter.requests[1:]
   }

   // 检查请求数是否超过阈值
   if len(limiter.requests) >= limiter.maxRequests {
      return false
   }

   limiter.requests = append(limiter.requests, currentTime)
   return true
}

func main() {
   limiter := NewSlidingWindowLimiter(500*time.Millisecond, 2) // 每秒最多允许4个请求
   for i := 0; i < 15; i++ {
      now := time.Now().Format("15:04:05")
      if limiter.AllowRequest() {
         fmt.Println(now + " 请求通过")
      } else {
         fmt.Println(now + " 请求被限流")
      }
      time.Sleep(100 * time.Millisecond)
   }
}
```


## 漏桶限流
```go
package main

import (
   "fmt"
   "time"
)

type LeakyBucket struct {
   rate       float64 // 漏桶速率，单位请求数/秒
   capacity   int     // 漏桶容量，最多可存储请求数
   water      int     // 当前水量，表示当前漏桶中的请求数
   lastLeakMs int64   // 上次漏水的时间戳，单位秒
}

func NewLeakyBucket(rate float64, capacity int) *LeakyBucket {
   return &LeakyBucket{
      rate:       rate,
      capacity:   capacity,
      water:      0,
      lastLeakMs: time.Now().Unix(),
   }
}

func (lb *LeakyBucket) Allow() bool {
   now := time.Now().Unix()
   elapsed := now - lb.lastLeakMs

   // 漏水，根据时间间隔计算漏掉的水量
   leakAmount := int(float64(elapsed) / 1000 * lb.rate) // 得出在elapsed的时间内应该有的最大请求数
   if leakAmount > 0 {
      if leakAmount > lb.water { // 当预估的最大请求数 > 已处理请求数
         lb.water = 0  // 表明了已经进来的请求数都被处理完了，可以直接置0
      } else {
         lb.water -= leakAmount
      }
   }

   // 判断当前水量是否超过容量
   if lb.water > lb.capacity {
      lb.water-- // 如果超过容量，减去刚刚增加的水量
      return false
   }

   // 增加水量
   lb.water++

   lb.lastLeakMs = now
   return true
}

func main() {
   // 创建一个漏桶，速率为每秒处理3个请求，容量为4个请求
   leakyBucket := NewLeakyBucket(3, 4)

   // 模拟请求
   for i := 1; i <= 15; i++ {
      now := time.Now().Format("15:04:05")
      if leakyBucket.Allow() {
         fmt.Printf(now+"  第 %d 个请求通过\n", i)
      } else {
         fmt.Printf(now+"  第 %d 个请求被限流\n", i)
      }
      time.Sleep(200 * time.Millisecond) // 模拟请求间隔
   }
}
```


## 令牌桶
```go
package main

import (
   "fmt"
   "sync"
   "time"
)

// TokenBucket 表示一个令牌桶。
type TokenBucket struct {
   rate       float64    // 令牌添加到桶中的速率。
   capacity   float64    // 桶的最大容量。
   tokens     float64    // 当前桶中的令牌数量。
   lastUpdate time.Time  // 上次更新令牌数量的时间。
   mu         sync.Mutex // 互斥锁，确保线程安全。
}

// NewTokenBucket 创建一个新的令牌桶，给定令牌添加速率和桶的容量。
func NewTokenBucket(rate float64, capacity float64) *TokenBucket {
   return &TokenBucket{
      rate:       rate,
      capacity:   capacity,
      tokens:     capacity, // 初始时，桶是满的。
      lastUpdate: time.Now(),
   }
}

// Allow 检查是否可以从桶中移除一个令牌。如果可以，它移除一个令牌并返回 true。
// 如果不可以，它返回 false。
func (tb *TokenBucket) Allow() bool {
   tb.mu.Lock()
   defer tb.mu.Unlock()

   // 根据经过的时间计算要添加的令牌数量。
   now := time.Now()
   elapsed := now.Sub(tb.lastUpdate).Seconds() 
   tokensToAdd := elapsed * tb.rate            

   tb.tokens += tokensToAdd
   if tb.tokens > tb.capacity {
      tb.tokens = tb.capacity // 确保令牌数量不超过桶的容量。
   }

   if tb.tokens >= 1.0 {
      tb.tokens--
      tb.lastUpdate = now
      return true
   }

   return false
}

func main() {
   tokenBucket := NewTokenBucket(2.0, 3.0)

   for i := 1; i <= 10; i++ {
      now := time.Now().Format("15:04:05")
      if tokenBucket.Allow() {
         fmt.Printf(now+"  第 %d 个请求通过\n", i)
      } else { // 如果不能移除一个令牌，请求被拒绝。
         fmt.Printf(now+"  第 %d 个请求被限流\n", i)
      }
      time.Sleep(200 * time.Millisecond)
   }
}
```

## 基于 redis的实现
```go
package main

import (
   "context"
   "fmt"
   "go.uber.org/atomic"
   "sync"

   "git.code.oa.com/pcg-csd/trpc-ext/redis"
)

type RedisClient interface {
   Do(ctx context.Context, cmd string, args ...interface{}) (interface{}, error)
}

// Client 数据库
type Client struct {
   client RedisClient // redis 操作
   script string      // lua脚本
}

// NewBucketClient 创建redis令牌桶
func NewBucketClient(redis RedisClient) *Client {
   helper := redis
   return &Client{
      client: helper,
      script: `
         -- 令牌桶限流脚本
         -- KEYS[1]: 桶的名称
         -- ARGV[1]: 桶的容量
         -- ARGV[2]: 令牌产生速率
         
         local bucket = KEYS[1]
         local capacity = tonumber(ARGV[1])
         local tokenRate = tonumber(ARGV[2])
         
         local redisTime = redis.call('TIME')
         local now = tonumber(redisTime[1])
         
         local tokens, lastRefill = unpack(redis.call('hmget', bucket, 'tokens', 'lastRefill'))
         tokens = tonumber(tokens)
         lastRefill = tonumber(lastRefill)
         
         if not tokens or not lastRefill then
            tokens = capacity
            lastRefill = now
         else
            local intervalsSinceLast = (now - lastRefill) * tokenRate
            tokens = math.min(capacity, tokens + intervalsSinceLast)
         end
         
         if tokens < 1 then
            return 0
         else
            redis.call('hmset', bucket, 'tokens', tokens - 1, 'lastRefill', now)
            return 1
         end
      `,
   }
}

// 获取令牌，获取成功则立即返回true，否则返回false
func (c *Client) isAllowed(ctx context.Context, key string, capacity int64, tokenRate int64) (bool, error) {
   result, err := redis.Int(c.client.Do(ctx, "eval", c.script, 1, key, capacity, tokenRate))
   if err != nil {
      fmt.Println("Redis 执行错误:", err)
      return false, err
   }
   return result == 1, nil
}

// 调用检测
func main() {
   c := NewBucketClient(redis.GetPoolByName("redis://127.0.0.1:6379"))
   gw := sync.WaitGroup{}
   gw.Add(120)
   count := atomic.Int64{}
   for i := 0; i < 120; i++ {
      go func(i int) {
         defer gw.Done()
         status, err := c.isAllowed(context.Background(), "test", 100, 10)
         if status {
            count.Add(1)
         }
         fmt.Printf("go %d status:%v error: %v\n", i, status, err)
      }(i)
   }
   gw.Wait()
   fmt.Printf("allow %d\n\n", count.Load())
}
```

## 基于etcd的实现
```go
package main

import (
    "context"
    "encoding/json"
    "fmt"
    "log"
    "net/http"
    "strconv"
    "sync"
    "time"

    "github.com/gin-gonic/gin"
    "go.etcd.io/etcd/client/v3"
    "golang.org/x/time/rate"
)

// 重试配置
type RetryConfig struct {
    MaxRetries    int           `json:"max_retries"`
    InitialDelay  time.Duration `json:"initial_delay"`
    MaxDelay      time.Duration `json:"max_delay"`
    BackoffFactor float64       `json:"backoff_factor"`
}

// 默认重试配置
var DefaultRetryConfig = RetryConfig{
    MaxRetries:    3,
    InitialDelay:  100 * time.Millisecond,
    MaxDelay:      1 * time.Second,
    BackoffFactor: 2.0,
}

// TokenBucket 令牌桶结构
type TokenBucket struct {
    Key       string  `json:"key"`
    Rate      float64 `json:"rate"`
    Capacity  int64   `json:"capacity"`
    Tokens    int64   `json:"tokens"`
    LastTime  int64   `json:"last_time"`
}

// DistributedRateLimiter 分布式限流器
type DistributedRateLimiter struct {
    client      *clientv3.Client
    localBuckets map[string]*LocalTokenBucket
    mutex       sync.RWMutex
    retryConfig RetryConfig
    fallbackStrategy FallbackStrategy
}

// LocalTokenBucket 本地缓存的令牌桶
type LocalTokenBucket struct {
    bucket    *TokenBucket
    lastSync  time.Time
    mutex     sync.Mutex
    lastTokens int64 // 用于降级时的本地令牌桶
}

// 降级策略
type FallbackStrategy int

const (
    FailOpen  FallbackStrategy = iota // 失败时开放（允许所有请求）
    FailClosed                        // 失败时关闭（拒绝所有请求）
    LocalBucket                       // 使用本地令牌桶
)

// 重试函数
func retry(ctx context.Context, config RetryConfig, operation func() error) error {
    var lastErr error
    
    delay := config.InitialDelay
    
    for i := 0; i <= config.MaxRetries; i++ {
        // 检查上下文是否已取消
        select {
        case <-ctx.Done():
            return ctx.Err()
        default:
        }
        
        err := operation()
        if err == nil {
            return nil // 成功
        }
        
        lastErr = err
        
        // 如果是最后一次重试，直接返回错误
        if i == config.MaxRetries {
            break
        }
        
        // 指数退避延迟
        select {
        case <-time.After(delay):
        case <-ctx.Done():
            return ctx.Err()
        }
        
        // 增加延迟时间
        delay = time.Duration(float64(delay) * config.BackoffFactor)
        if delay > config.MaxDelay {
            delay = config.MaxDelay
        }
    }
    
    return fmt.Errorf("operation failed after %d retries: %w", config.MaxRetries, lastErr)
}

// 创建分布式限流器
func NewDistributedRateLimiter(endpoints []string, retryConfig RetryConfig, fallbackStrategy FallbackStrategy) (*DistributedRateLimiter, error) {
    cli, err := clientv3.New(clientv3.Config{
        Endpoints:   endpoints,
        DialTimeout: 5 * time.Second,
    })
    if err != nil {
        return nil, err
    }

    return &DistributedRateLimiter{
        client:           cli,
        localBuckets:     make(map[string]*LocalTokenBucket),
        retryConfig:      retryConfig,
        fallbackStrategy: fallbackStrategy,
    }, nil
}

// 带重试的创建令牌桶
func (drl *DistributedRateLimiter) CreateBucket(ctx context.Context, key string, rate float64, capacity int64) error {
    return retry(ctx, drl.retryConfig, func() error {
        bucket := &TokenBucket{
            Key:      key,
            Rate:     rate,
            Capacity: capacity,
            Tokens:   capacity,
            LastTime: time.Now().Unix(),
        }

        data, err := json.Marshal(bucket)
        if err != nil {
            return err
        }

        _, err = drl.client.Put(ctx, fmt.Sprintf("/ratelimit/%s", key), string(data))
        return err
    })
}

// 带重试和降级的获取令牌
func (drl *DistributedRateLimiter) Allow(ctx context.Context, key string) (bool, error) {
    // 尝试从分布式令牌桶获取令牌
    allowed, err := drl.allowWithRetry(ctx, key)
    if err == nil {
        return allowed, nil
    }

    // 分布式令牌桶失败，执行降级策略
    log.Printf("Distributed rate limit failed for key %s: %v, applying fallback strategy", key, err)
    return drl.applyFallbackStrategy(ctx, key, err)
}

// 带重试的获取令牌
func (drl *DistributedRateLimiter) allowWithRetry(ctx context.Context, key string) (bool, error) {
    var result bool
    var lastErr error

    err := retry(ctx, drl.retryConfig, func() error {
        allowed, err := drl.allowOnce(ctx, key)
        if err != nil {
            lastErr = err
            return err
        }
        result = allowed
        return nil
    })

    if err != nil {
        return false, fmt.Errorf("failed to get tokens after retries: %w", lastErr)
    }

    return result, nil
}

// 单次获取令牌操作
func (drl *DistributedRateLimiter) allowOnce(ctx context.Context, key string) (bool, error) {
    localBucket := drl.getLocalBucket(key)
    localBucket.mutex.Lock()
    defer localBucket.mutex.Unlock()

    now := time.Now().Unix()

    // 从 etcd 获取最新的桶状态
    resp, err := drl.client.Get(ctx, fmt.Sprintf("/ratelimit/%s", key))
    if err != nil {
        return false, fmt.Errorf("failed to get bucket from etcd: %w", err)
    }

    if len(resp.Kvs) == 0 {
        return false, fmt.Errorf("bucket not found: %s", key)
    }

    var bucket TokenBucket
    if err := json.Unmarshal(resp.Kvs[0].Value, &bucket); err != nil {
        return false, fmt.Errorf("failed to unmarshal bucket: %w", err)
    }

    // 计算应该添加的令牌数
    elapsed := float64(now - bucket.LastTime)
    newTokens := int64(elapsed * bucket.Rate)
    
    // 更新桶状态
    bucket.Tokens += newTokens
    if bucket.Tokens > bucket.Capacity {
        bucket.Tokens = bucket.Capacity
    }
    bucket.LastTime = now

    // 尝试获取令牌
    if bucket.Tokens > 0 {
        bucket.Tokens--
        
        // 更新到 etcd
        data, err := json.Marshal(bucket)
        if err != nil {
            return false, fmt.Errorf("failed to marshal bucket: %w", err)
        }

        // 使用事务确保原子性
        txnResp, err := drl.client.Txn(ctx).
            If(clientv3.Compare(clientv3.Value(fmt.Sprintf("/ratelimit/%s", key)), "=", string(resp.Kvs[0].Value))).
            Then(clientv3.OpPut(fmt.Sprintf("/ratelimit/%s", key), string(data))).
            Commit()
        
        if err != nil {
            return false, fmt.Errorf("failed to commit transaction: %w", err)
        }

        if !txnResp.Succeeded {
            return false, fmt.Errorf("transaction failed, concurrent update detected")
        }

        // 更新本地缓存
        localBucket.bucket = &bucket
        localBucket.lastTokens = bucket.Tokens // 更新本地令牌数
        return true, nil
    }

    // 没有足够令牌
    localBucket.bucket = &bucket
    localBucket.lastTokens = bucket.Tokens
    return false, nil
}

// 应用降级策略
func (drl *DistributedRateLimiter) applyFallbackStrategy(ctx context.Context, key string, err error) (bool, error) {
    switch drl.fallbackStrategy {
    case FailOpen:
        // 失败时开放，允许所有请求
        log.Printf("FailOpen strategy: allowing request for key %s", key)
        return true, nil
        
    case FailClosed:
        // 失败时关闭，拒绝所有请求
        log.Printf("FailClosed strategy: rejecting request for key %s", key)
        return false, fmt.Errorf("rate limit service unavailable: %w", err)
        
    case LocalBucket:
        // 使用本地令牌桶
        return drl.allowWithLocalBucket(ctx, key)
        
    default:
        return false, fmt.Errorf("unknown fallback strategy: %d", drl.fallbackStrategy)
    }
}

// 使用本地令牌桶（降级方案）
func (drl *DistributedRateLimiter) allowWithLocalBucket(ctx context.Context, key string) (bool, error) {
    localBucket := drl.getLocalBucket(key)
    localBucket.mutex.Lock()
    defer localBucket.mutex.Unlock()

    // 如果本地没有桶配置，创建一个默认的
    if localBucket.bucket == nil {
        localBucket.bucket = &TokenBucket{
            Key:      key,
            Rate:     10, // 默认每秒10个令牌
            Capacity: 20, // 默认容量20
            Tokens:   20, // 初始满的
            LastTime: time.Now().Unix(),
        }
        localBucket.lastTokens = 20
    }

    now := time.Now().Unix()
    bucket := localBucket.bucket

    // 计算本地令牌数（基于时间的令牌桶算法）
    elapsed := float64(now - bucket.LastTime)
    newTokens := int64(elapsed * bucket.Rate)
    
    localTokens := localBucket.lastTokens + newTokens
    if localTokens > bucket.Capacity {
        localTokens = bucket.Capacity
    }

    if localTokens > 0 {
        localTokens--
        localBucket.lastTokens = localTokens
        bucket.LastTime = now
        return true, nil
    }

    localBucket.lastTokens = localTokens
    bucket.LastTime = now
    return false, nil
}

// 获取本地缓存的桶
func (drl *DistributedRateLimiter) getLocalBucket(key string) *LocalTokenBucket {
    drl.mutex.Lock()
    defer drl.mutex.Unlock()

    if localBucket, exists := drl.localBuckets[key]; exists {
        return localBucket
    }

    localBucket := &LocalTokenBucket{
        bucket: &TokenBucket{Key: key},
        lastTokens: 0,
    }
    drl.localBuckets[key] = localBucket
    return localBucket
}

// 健康检查
func (drl *DistributedRateLimiter) HealthCheck(ctx context.Context) error {
    return retry(ctx, drl.retryConfig, func() error {
        _, err := drl.client.Get(ctx, "/health")
        return err
    })
}

// HTTP 服务示例
type RateLimitService struct {
    limiter *DistributedRateLimiter
}

func NewRateLimitService(limiter *DistributedRateLimiter) *RateLimitService {
    return &RateLimitService{
        limiter: limiter,
    }
}

func (rls *RateLimitService) RateLimitMiddleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        userID := c.GetHeader("X-User-ID")
        if userID == "" {
            userID = c.ClientIP()
        }

        bucketKey := fmt.Sprintf("user:%s", userID)
        
        ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
        defer cancel()

        allowed, err := rls.limiter.Allow(ctx, bucketKey)
        if err != nil {
            log.Printf("Rate limit error for user %s: %v", userID, err)
            // 根据降级策略，可能仍然允许请求
            if !allowed {
                c.JSON(http.StatusTooManyRequests, gin.H{
                    "error": "Rate limit exceeded or service unavailable",
                    "message": err.Error(),
                    "user_id": userID,
                })
                c.Abort()
                return
            }
            // 降级策略允许请求，继续处理但记录警告
            log.Printf("Request allowed due to fallback strategy for user %s", userID)
        }

        if !allowed {
            c.JSON(http.StatusTooManyRequests, gin.H{
                "error": "Rate limit exceeded",
                "message": "Too many requests, please try again later",
                "user_id": userID,
            })
            c.Abort()
            return
        }

        c.Next()
    }
}

func (rls *RateLimitService) SetupRoutes(r *gin.Engine) {
    // 健康检查接口
    r.GET("/health", func(c *gin.Context) {
        ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
        defer cancel()
        
        if err := rls.limiter.HealthCheck(ctx); err != nil {
            c.JSON(http.StatusServiceUnavailable, gin.H{
                "status": "unhealthy",
                "error": err.Error(),
            })
            return
        }
        
        c.JSON(http.StatusOK, gin.H{
            "status": "healthy",
        })
    })

    // 创建限流桶
    r.POST("/admin/bucket", func(c *gin.Context) {
        key := c.Query("key")
        rateStr := c.Query("rate")
        capacityStr := c.Query("capacity")

        rate, err := strconv.ParseFloat(rateStr, 64)
        if err != nil {
            c.JSON(http.StatusBadRequest, gin.H{"error": "Invalid rate"})
            return
        }

        capacity, err := strconv.ParseInt(capacityStr, 10, 64)
        if err != nil {
            c.JSON(http.StatusBadRequest, gin.H{"error": "Invalid capacity"})
            return
        }

        ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
        defer cancel()

        if err := rls.limiter.CreateBucket(ctx, key, rate, capacity); err != nil {
            c.JSON(http.StatusInternalServerError, gin.H{
                "error": "Failed to create bucket",
                "message": err.Error(),
            })
            return
        }

        c.JSON(http.StatusOK, gin.H{"message": "Bucket created successfully"})
    })

    // 受限流保护的业务接口
    protected := r.Group("/api")
    protected.Use(rls.RateLimitMiddleware())
    {
        protected.GET("/data", func(c *gin.Context) {
            c.JSON(http.StatusOK, gin.H{
                "message": "Protected data",
                "timestamp": time.Now().Unix(),
                "user_id": c.GetHeader("X-User-ID"),
            })
        })
    }
}

func main() {
    // 连接 etcd，配置重试和降级策略
    limiter, err := NewDistributedRateLimiter(
        []string{"localhost:2379"},
        DefaultRetryConfig,
        LocalBucket, // 使用本地令牌桶作为降级策略
    )
    if err != nil {
        log.Fatal("Failed to connect to etcd:", err)
    }
    defer limiter.client.Close()

    // 预创建一些限流桶
    ctx := context.Background()
    limiter.CreateBucket(ctx, "user:default", 10, 20)

    // 创建 HTTP 服务
    r := gin.Default()
    service := NewRateLimitService(limiter)
    service.SetupRoutes(r)

    log.Println("Server starting on :8080 with distributed rate limiting")
    if err := r.Run(":8080"); err != nil {
        log.Fatal("Failed to start server:", err)
    }
}
```
